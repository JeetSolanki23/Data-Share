# Data Share - Deployment Commands

## Quick Start - Local Development
python main.py

## Production Deployment - Gunicorn (VPS, Azure, etc.)
gunicorn -w 4 -b 0.0.0.0:5000 --timeout 300 main:app

## Production Deployment - Gunicorn with Systemd
# See DEPLOYMENT.md for systemd service setup
systemctl start data-share
systemctl status data-share

## Docker Deployment
docker build -t data-share:1.0 .
docker run -d -p 5000:5000 \
  -v $(pwd)/storage:/app/storage \
  -e SECRET_KEY=<your-key> \
  -e ENVIRONMENT=production \
  data-share:1.0

## Docker Compose (with PostgreSQL)
docker-compose up -d

## Azure App Service Deployment
az webapp create --resource-group <group> --plan <plan> --name <app-name> --runtime "PYTHON:3.11"
git push azure main

## Vercel Serverless Deployment
# Set environment variables in Vercel dashboard:
# - SECRET_KEY
# - ENVIRONMENT=production
# - DATABASE_URL=postgresql://...
# - CLOUDINARY_URL=cloudinary://...
# Then deploy: git push

## Heroku Deployment
heroku create <app-name>
git push heroku main
heroku config:set SECRET_KEY=<your-key>

## Render Deployment
# Connect GitHub repo in Render dashboard
# Build command: pip install -r requirements.txt
# Start command: gunicorn -w 4 main:app

## AWS EC2 Deployment
ssh -i key.pem ubuntu@<instance-ip>
# Follow VPS setup in DEPLOYMENT.md

## Google Cloud Run Deployment
gcloud run deploy data-share \
  --source . \
  --platform managed \
  --region us-central1
